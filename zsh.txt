# Fetch every remote and remove local branches whose upstream branch has
# already been deleted. Branches that have not been merged are kept by -d.
#
# The currently checked-out branch will never be removed.
unalias git-prune-local 2>/dev/null || true
git-prune-local() {
  local current_branch branch upstream

  git fetch --all --prune || return
  current_branch="$(git branch --show-current)" || return

  while IFS=$'\t' read -r branch upstream; do
    [[ -n "$upstream" && "$branch" != "$current_branch" ]] || continue
    git show-ref --verify --quiet "$upstream" && continue
    git branch -d -- "$branch"
  done < <(
    git for-each-ref \
      --format=$'%(refname:short)\t%(upstream)' \
      refs/heads
  )
}
