# --------------------------------------------------
# Git
# --------------------------------------------------

# Fetch remote branches and remove local branches whose upstream
# branch has already been deleted.
#
# The currently checked-out branch will never be removed.
alias git-prune-local='git fetch -p && current_branch=$(git branch --show-current) && for branch in $(git branch -vv | grep ": gone]" | awk "{print \$1}"); do [ "$branch" != "$current_branch" ] && git branch -d "$branch"; done'
